\begin{algorithm*}[t]
\caption{Retrospective candidate construction and PEC ranking}
\label{alg:pec-ranking}
\small
\begin{algorithmic}[1]
\Require Processed profiles $S$, articles $A$, clicks $L$, traces $R$; queries $\mathcal Q$; regime $g$; frozen popularity $v$
\Require Frozen profile/article embeddings; selected and refitted scorer $f_\theta$; ordered feature schema $\mathcal F$
\Ensure Ranked lists $\{\pi_q\}$ and query-level evaluation metrics $\{m_q\}$
\For{each $q=(u,t)\in\mathcal Q$}
  \State $Y_q\gets$ articles clicked by $u$ on $t$; $R_q\gets$ retained items with valid ranks for $(u,t)$
  \State $R_q^-\gets$ sample up to 24 items from $R_q\setminus Y_q$
  \State $K_q\gets$ articles in window $[t-3,t]$ if $g=\mathrm{recent}$, otherwise $[t,t]$
  \State $b_q\gets\min(24,\max(4,4|Y_q|))$; $K_q\gets K_q\setminus(Y_q\cup R_q)$
  \State $N_q\gets\operatorname{Select}_g(K_q,b_q;v)$ \Comment{Random sampling or frozen-popularity/file-order selection}
  \State $\mathcal A_q\gets\operatorname{unique}(Y_q\cup R_q^-\cup N_q)$
  \State $H_q^{\mathrm{count}}\gets$ all eligible clicks by $u$ strictly before $t$; $H_q^{\mathrm{sem}}\gets$ their latest 50 events
  \For{each candidate $i\in\mathcal A_q$}
    \State Construct $P_{u,i},\mathbf E_{q,i},\mathbf C_{q,i},\mathbf z_{q,i}$ from the corresponding sources
    \State $\boldsymbol\psi_{q,i}\gets$ six specified source products; $\mathbf x_{q,i}\gets\operatorname{Serialize}(P,\mathbf E,\mathbf C,\mathbf z,\boldsymbol\psi;\mathcal F)$
    \State $s_{q,i}\gets f_\theta(\mathbf x_{q,i})$
  \EndFor
  \State $\pi_q\gets\operatorname{argsort}^{\downarrow}_{i\in\mathcal A_q}s_{q,i}$; $m_q\gets\operatorname{Metrics}(\pi_q,Y_q)$
\EndFor
\State \Return $\{\pi_q,m_q:q\in\mathcal Q\}$
\end{algorithmic}
\vspace{2pt}
\footnotesize $\operatorname{Select}_g$ samples without replacement in recent-window; same-day sorts by $v$ and breaks ties by source-file order. Budgets are capped by available items. Positives define this retrospective evaluation pool, not a prospective retrieval procedure. User-cluster uncertainty is computed from the returned query metrics.
\end{algorithm*}
